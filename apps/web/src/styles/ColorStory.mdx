export const TextSample = ({ textToken, bgToken, label }) => {
  const textValue = useTokenValue(textToken);
  const bgValue = useTokenValue(bgToken);
  const ratio = contrastRatio(textValue, bgValue);
  return (
    <div style={{
      display: 'flex', alignItems: 'center', justifyContent: 'space-between',
      padding: '12px 16px', marginBottom: '8px', borderRadius: '6px',
      background: `var(${bgToken})`, border: '1px solid var(--color-border)',
    }}>
      <span style={{ color: `var(${textToken})`, fontSize: '15px' }}>{label}</span>
      <span style={{ fontFamily: 'var(--font-mono)', fontSize: '12px', color: `var(${textToken})`, opacity: 0.75 }}>
        {ratio ? `${ratio.toFixed(2)}:1` : '—'}
      </span>
    </div>
  );
};

# Color Story

A cleared, uncluttered ground; a single confident accent that means *the one next thing*. There is no second accent — every other color on the screen (status, disabled state, the slide tints) is deliberately quiet, so the one place color is used to mean "act here" keeps meaning it. Switch modes with the **Preview** toggle below — every swatch reads its value live from the current cascade, so this page cannot drift from what the app actually renders.

## Do / Don't

- The accent is reserved for the single active priority on screen — a button, a selected row, a focus ring. Never decoration.
- Never place two accented elements on one screen competing for attention. If something else needs emphasis, it should not be the accent color.
- Disabled is darker than surface, not lighter — on a warm ground, a lighter "disabled" would read as *brighter*, the opposite of inactive.
